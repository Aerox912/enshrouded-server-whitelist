cmake_minimum_required(VERSION 3.20)
project(esm_admission LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/W4> $<$<COMPILE_LANGUAGE:CXX>:/WX> $<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
add_library(admission_policy STATIC policy.cpp)
add_executable(admission_policy_tests policy_tests.cpp)
target_link_libraries(admission_policy_tests PRIVATE admission_policy)
enable_testing()
add_test(NAME admission_policy COMMAND admission_policy_tests)
add_executable(admission_launch_fixture launch_fixture.cpp)
add_library(admission_guard_fixture SHARED guard_fixture.cpp)
add_executable(admission_launcher_tests_host launcher.cpp)
target_compile_definitions(admission_launcher_tests_host PRIVATE ESM_ADMISSION_LAUNCH_TESTING)
target_link_libraries(admission_launcher_tests_host PRIVATE bcrypt)
add_executable(esm_admission_launcher launcher.cpp)
target_link_libraries(esm_admission_launcher PRIVATE bcrypt)
add_library(admission_runtime STATIC runtime.cpp)
target_link_libraries(admission_runtime PUBLIC admission_policy bcrypt)
add_executable(admission_native_tests native_tests.cpp)
target_link_libraries(admission_native_tests PRIVATE admission_runtime)
add_library(esm_admission SHARED dll.cpp)
target_link_libraries(esm_admission PRIVATE admission_runtime)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/guard_artifact.hpp"
 COMMAND "${CMAKE_COMMAND}" "-DINPUT=$<TARGET_FILE:esm_admission>" "-DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/guard_artifact.hpp" -P "${CMAKE_CURRENT_SOURCE_DIR}/pin_guard.cmake"
 DEPENDS esm_admission "${CMAKE_CURRENT_SOURCE_DIR}/pin_guard.cmake" VERBATIM)
target_sources(esm_admission_launcher PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/guard_artifact.hpp")
target_include_directories(esm_admission_launcher PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
foreach(target admission_runtime esm_admission esm_admission_launcher)
 target_compile_options(${target} PRIVATE /guard:cf)
 target_link_options(${target} PRIVATE /GUARD:CF /DYNAMICBASE)
endforeach()
set(ADMISSION_SERVER_FIXTURE "" CACHE FILEPATH "Pinned current dedicated PE")
set(ADMISSION_PYTHON "" CACHE FILEPATH "Python with native analysis packages")
if(ADMISSION_SERVER_FIXTURE AND ADMISSION_PYTHON)
 foreach(mode native tamper invalid no-cleanup)
  if(mode STREQUAL native)
   add_test(NAME admission_native COMMAND admission_native_tests "${ADMISSION_SERVER_FIXTURE}")
  else()
   add_test(NAME admission_${mode} COMMAND admission_native_tests "${ADMISSION_SERVER_FIXTURE}" ${mode})
  endif()
 endforeach()
 add_test(NAME admission_launcher COMMAND "${ADMISSION_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/launcher_tests.py" "$<TARGET_FILE_DIR:admission_launcher_tests_host>")
 add_test(NAME admission_native_removal COMMAND "${ADMISSION_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/native_removal_test.py" "${ADMISSION_SERVER_FIXTURE}")
 add_test(NAME admission_artifact_pin COMMAND "${ADMISSION_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/artifact_test.py" "$<TARGET_FILE_DIR:esm_admission_launcher>" "${ADMISSION_SERVER_FIXTURE}")
endif()
